# Raspberry Pi 3 Model B+ (milestone 9 of docs/rpi3-plan.md, without the H.264 decoder itself):
# VCHIQ and the VideoCore firmware's services the stock image's drivers use. The image boots
# unmodified (dtparam=audio=on in its config.txt turns on the headphone output); after the
# first-boot wizard, logged in on the serial console:
# - the media drivers bound over VCHIQ: bcm2835-codec's decoder, encoder, ISP, image-effects
#   and JPEG devices, bcm2835-isp's nodes; vc-sm-cma's /dev/vcsm-cma;
# - the decoder through V4L2 with v4l2-ctl: H.264 in (a 320x240 stream of parameter sets and
#   empty slices written with printf), YU12 out; the stand-in decoder turns each coded
#   picture into colour bars, so the first luma byte is the bars' white (180, 0xb4);
# - the ISP: a red 64x32 RGB24 picture scaled to 32x16 YU12 (BT.601 red: Y 82 = 0x52);
# - the headphone card (bcm2835 Headphones): a 440 Hz sine through the firmware's audio
#   service, written to --audio-out on the host.
#   sbc-emu run --board pi3b --sd cache/os/raspios-lite.img --audio-out out/rpi3-vchiq.wav \
#       --script scripts/rpi3-vchiq.txt --timeout 900
wait-serial raspberrypi login:
# the first-boot wizard on HDMI, answered with the USB keyboard
wait-pixel 10 700 0000aa
wait-idle 3
type \n
sleep 6
type pi\n
sleep 6
type raspberry\n
sleep 6
type raspberry\n
sleep 8
type \n
sleep 10
serial \r
wait-serial raspberrypi login:
serial pi\r
wait-serial Password:
serial raspberry\r
wait-serial pi@raspberrypi
serial sudo -k; sudo -s\r
wait-serial password for pi
serial raspberry\r
wait-serial root@raspberrypi
# the drivers over VCHIQ, and nothing failed
serial dmesg | grep --color=never -E 'bcm2835-codec|bcm2835-isp|vc_sm|vcsm|snd_bcm2835|mmal|vchiq'; echo DMESG_$((1))\r
wait-serial DMESG_1
serial echo FAIL_$(dmesg | grep -ciE 'oops|vchiq.*(fail|error)|mmal.*(fail|error)|codec.*fail')_END\r
wait-serial FAIL_0_END
serial v4l2-ctl --list-devices; echo LIST_$(v4l2-ctl --list-devices | grep -A1 --color=never bcm2835-codec-decode | tail -1 | tr -d '[:space:]')_$(ls /dev/vcsm-cma)_END\r
wait-serial LIST_/dev/video10_/dev/vcsm-cma_END
# the decoder's input formats: H.264 (no MPEG-2 or VC-1: no licence key)
serial v4l2-ctl -d /dev/video10 --list-formats-out | grep --color=never -E "'"; echo FMT_$(v4l2-ctl -d /dev/video10 --list-formats-out | grep -c -E "'(H264|MPG2|WVC1)'")_END\r
wait-serial FMT_1_END
# decode: four coded pictures in, YU12 out through the stand-in decoder, until the end of the
# stream (v4l2-ctl stops the decoder at the end of its file and stops at the last buffer)
serial printf '\x00\x00\x00\x01\x67\x42\x00\x1e\xf4\x0a\x0f\xc8\x00\x00\x00\x01\x65\x88\xc0\x00\x00\x00\x01\x41\x9b\x00\x00\x00\x01\x41\x9b\x00\x00\x00\x01\x41\x9b' > /tmp/t.h264; echo H264_$(stat -c %s /tmp/t.h264)_END\r
wait-serial H264_37_END
serial v4l2-ctl -d /dev/video10 --set-fmt-video-out=pixelformat=H264 --set-fmt-video=width=320,height=240,pixelformat=YU12 --stream-mmap --stream-out-mmap --stream-from=/tmp/t.h264 --stream-to=/tmp/out.yuv 2>&1 | tr -d '<>' | grep --color=never -E 'EVENT|STOP|rror'; echo DEC_$(stat -c %s /tmp/out.yuv)_$(od -An -tx1 -N1 /tmp/out.yuv | tr -d ' ')_END\r
wait-serial DEC_460800_b4_END
# the ISP: red RGB24 64x32 -> YU12 32x16
serial for i in $(seq 2048); do printf '\377\0\0'; done > /tmp/red.rgb; v4l2-ctl -d /dev/video12 --set-fmt-video-out=width=64,height=32,pixelformat=RGB3 --set-fmt-video=width=32,height=16,pixelformat=YU12 --stream-mmap --stream-out-mmap --stream-from=/tmp/red.rgb --stream-to=/tmp/isp.yuv --stream-count=1 2>&1 | tail -2; echo ISP_$(od -An -tx1 -N1 /tmp/isp.yuv | tr -d ' ')_END\r
wait-serial ISP_52_END
# the headphone jack: 440 Hz through the firmware's audio service
serial aplay -l | grep --color=never -E '^card'; echo CARDS_$(aplay -l | grep -c 'bcm2835 Headphones')_END\r
wait-serial CARDS_1_END
serial speaker-test -D plughw:CARD=Headphones,DEV=0 -c 2 -r 48000 -t sine -f 440 -l 1 2>&1 | grep --color=never -E 'Rate|Front|rror'; echo AUDIO_$((5))\r
wait-serial AUDIO_5
sleep 2
quit
