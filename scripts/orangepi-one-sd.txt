# Orange Pi One SD card (sunxi-mmc with IDMAC): an ext4 card made on the host, read,
# written, verified after dropping the page cache (so data comes from the device), trimmed
# (erase commands), and read again after a guest reboot.
# Card: mkfs.ext4 -d DIR card.img 64M with DIR/hello.txt = HELLO_FROM_HOST, then
#   sbc-emu run --board orangepi-one ... --sd card.img --script scripts/orangepi-one-sd.txt
wait-serial orangepi-one#
serial cat /sys/block/mmcblk0/size | sed 's/^/SECTORS_/'\r
wait-serial SECTORS_131072
serial modprobe ext4; mount -t ext4 /dev/mmcblk0 /mnt && cat /mnt/hello.txt\r
wait-serial HELLO_FROM_HOST
serial echo guest-was-here > /mnt/guest.txt; dd if=/dev/urandom of=/mnt/blob bs=64k count=64 2>/dev/null; sha256sum /mnt/blob > /mnt/blob.sha; sync; echo 3 > /proc/sys/vm/drop_caches; sha256sum -c /mnt/blob.sha | sed 's/^/CHECK_/'\r
wait-serial CHECK_/mnt/blob: OK
serial fstrim -v /mnt && echo TRIM_DONE\r
wait-serial TRIM_DONE
serial umount /mnt; reboot -f\r
wait-serial Booting Linux on physical CPU
wait-serial orangepi-one#
serial modprobe ext4; mount -t ext4 /dev/mmcblk0 /mnt && cat /mnt/guest.txt && sha256sum -c /mnt/blob.sha | sed 's/^/AFTER_REBOOT_/'\r
wait-serial AFTER_REBOOT_/mnt/blob: OK
serial umount /mnt; poweroff -f\r
