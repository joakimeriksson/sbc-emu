# Tegra234 xHCI with a USB drive and a webcam plugged in. Run with --usb-drive on an image whose
# first bytes are SBC_EMU_USB_DRIVE (docs/jetson-orin-nano.md, "USB") and --camera.
wait-serial TEGRA234_BOOT_OK
serial for i in $(seq 1 50); do [ -b /dev/sda ] && break; sleep 0.2; done; [ "$(head -c 17 /dev/sda)" = SBC_EMU_USB_DRIVE ] && echo "DRIVE_" "OK"\r
wait-serial DRIVE_ OK
serial grep -qx 'Virtual USB Camera' /sys/bus/usb/devices/*/product && echo "CAMERA_" "ENUMERATED"\r
wait-serial CAMERA_ ENUMERATED
quit
