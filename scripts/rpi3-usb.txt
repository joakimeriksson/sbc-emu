# Raspberry Pi 3 Model B+ (milestone 3 of docs/rpi3-plan.md): USB and Ethernet from the Pi OS
# kernel with a BusyBox initramfs (the kernel has `dwc2`, `lan78xx` and `usbhid` built in). The
# hub tree behind the DWC2 root port (the LAN7515's two high-speed hubs, its LAN7800, the
# full-speed keyboard and tablet through the hub's transaction translator), the gigabit link,
# the MAC address from the device tree, DHCP and ping over the virtual network, a DNS answer,
# and typing on tty1 with the keyboard.
#   sbc-emu run --board pi3b --kernel cache/rpi3/Image --initrd cache/rpi3/initramfs.cpio \
#     --script scripts/rpi3-usb.txt --timeout 180
wait-serial pi3#
serial sleep 3; for d in /sys/bus/usb/devices/[0-9]-*; do case $d in *:*) continue;; esac; printf '%s=%s:%s@%s ' ${d##*/} $(cat $d/idVendor $d/idProduct $d/speed); done; echo\r
wait-serial 1-1=0424:2514@480 1-1.1=0424:2514@480 1-1.1.1=0424:7800@480 1-1.2=1209:0001@12 1-1.3=1209:0002@12
serial ip link set eth0 up; sleep 3; echo $(cat /sys/class/net/eth0/address /sys/class/net/eth0/speed /sys/class/net/eth0/carrier) LINK\r
wait-serial 02:5b:c0:00:03:01 1000 1 LINK
# udhcpc hands the lease to a script; this one sets the address, the route and the resolver
serial printf '#!/bin/sh\012[ "$1" = bound ] || exit 0\012ip addr add $ip/$mask dev $interface\012ip route add default via $router\012echo nameserver $dns > /etc/resolv.conf\012' > /tmp/dhcp.sh; chmod +x /tmp/dhcp.sh\r
serial udhcpc -i eth0 -n -q -t 5 -s /tmp/dhcp.sh > /dev/null 2>&1; ip -4 addr show eth0 | grep inet\r
wait-serial inet 10.0.2.15/24
serial ping -c 3 $(ip route | awk '/default/ { print $3 }') | grep received\r
wait-serial 3 packets received
serial nslookup example.com > /dev/null && echo DNS_OK$((1))\r
wait-serial DNS_OK1
serial (read -t 20 line < /dev/tty1; echo "TYPED_[$line]") & sleep 1\r
type hello from the usb keyboard\n
wait-serial TYPED_[hello from the usb keyboard]
serial echo USB_$((2 * 21))_OK\r
wait-serial USB_42_OK
quit
