# Orange Pi RV GPU: Debian's vkcube on the HDMI output, rendered by the PowerVR BXE-4-32
# (mainline powervr + Mesa's PowerVR Vulkan driver) and presented through KMS (VK_KHR_display
# on simpledrm). Use --hdmi-mode 640x480: GPU work runs synchronously, so guest time stands
# still while frames render (about 3.5 min wall at 640x480). Needs scripts/orangepi-rv-gpu.sh.
wait-serial orangepi-rv#
serial modprobe powervr exp_hw_support=1; mount -t devtmpfs dev /debian/dev; mount -t proc proc /debian/proc; mount -t sysfs sys /debian/sys; echo MNT_""OK\n
wait-serial MNT_OK
serial (PVR_I_WANT_A_BROKEN_VULKAN_DRIVER=1 chroot /debian vkcube --wsi display > /tmp/cube.log 2>&1 &); sleep 1; grep 'Selected GPU' /tmp/cube.log; echo CUBE_""SHOT\n
wait-serial Selected GPU 0: PowerVR B-Series BXE-4-32 MC1
wait-serial CUBE_SHOT
screenshot out/orangepi-rv-vkcube.png
quit
