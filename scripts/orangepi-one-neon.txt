# NEON userland on the Orange Pi One: Linux reports HWCAP_NEON, and Debian's armhf glibc,
# OpenSSL and libjpeg-turbo take their Advanced SIMD paths. Each result is the one the
# host's native OpenSSL 3 / libjpeg-turbo 3 (and Python's hashlib) give for the same
# input, and the scalar paths (OPENSSL_armcap=0, JSIMD_FORCENONE=1) agree.
# Fixture: scripts/orangepi-one-fixture.sh and scripts/debian-armhf-neon-fixture.sh, then
#   sbc-emu run --board orangepi-one --kernel cache/orangepi-one/zImage \
#     --dtb cache/orangepi-one/sun8i-h3-orangepi-one.dtb \
#     --initrd cache/debian-armhf-neon/initramfs.cpio.gz --script scripts/orangepi-one-neon.txt
wait-serial neon#
serial grep -m1 -o -w neon /proc/cpuinfo | sed 's/^/HWCAP_/'\r
wait-serial HWCAP_neon
serial openssl version -c\r
wait-serial OPENSSL_armcap=0x1
serial seq 1 200000 > /tmp/d; openssl dgst -sha1 /tmp/d\r
wait-serial 17454322f38ec2b6b6b43587dee97fcabaf998b6
serial openssl dgst -sha256 /tmp/d\r
wait-serial 5af7b95208fdcff454bab3f5eddf567a688a3796c703d4fef91072e38645c062
serial openssl dgst -sha512 /tmp/d\r
wait-serial b5fd978b41dd6da3ce93ced1d2805ffd0f7e238fc75d06397972a475697adc24ef919f56e1101c99a1e3dcefffa6816a90cb724b7f8f46ecf4f75116ef2ca7e3
serial OPENSSL_armcap=0 openssl dgst -sha512 /tmp/d | sed 's/^/SCALAR_/'\r
wait-serial SCALAR_SHA2-512(/tmp/d)= b5fd978b41dd6da3ce93ced1d2805ffd0f7e238fc75d06397972a475697adc24ef919f56e1101c99a1e3dcefffa6816a90cb724b7f8f46ecf4f75116ef2ca7e3
serial openssl enc -chacha20 -K 000102030405060708090a0b0c0d0e0f101112131415161718191a1b1c1d1e1f -iv 00000000000000000000000000000001 -in /tmp/d | sha256sum\r
wait-serial 3b5fed4106583969104593c5b07b617ee6f00b67e307433b40c6380f47a5f2a8
serial openssl mac -macopt hexkey:85d6be7857556d337f4452fe42d506a80103808afb0db2fd4abff6af4149f51b -in /tmp/d poly1305\r
wait-serial 736437E6201A5B971595454AE07D6460
serial awk 'BEGIN{print "P3"; print "64 48"; print "255"; for(y=0;y<48;y++) for(x=0;x<64;x++) print (x*4)%256, (y*5)%256, (x*y)%256}' > /tmp/i.ppm; cjpeg -quality 90 /tmp/i.ppm > /tmp/i.jpg; md5sum /tmp/i.jpg\r
wait-serial 0524a22965b37a70af5bb5f49b1fc9f6
serial djpeg /tmp/i.jpg | md5sum\r
wait-serial 1c75c3c9b624aa1d2665ac937fbebfb7
serial JSIMD_FORCENONE=1 djpeg /tmp/i.jpg | md5sum | sed 's/^/SCALAR_/'\r
wait-serial SCALAR_1c75c3c9b624aa1d2665ac937fbebfb7
serial poweroff -f\r
