# Exercise Tegra234 UART A through Linux and the separate host UART loopback.
# Run with --rover-uart loopback and the Orin Alpine fixture.
wait-serial TEGRA234_BOOT_OK
serial exec 3<>/dev/ttyS1; stty raw -echo 115200 <&3 && echo "UART_" "READY"\r
wait-serial UART_ READY
serial printf %s OK >&3; dd bs=1 count=2 <&3 2>/dev/null | grep -qx OK && echo "SHORT_" "OK"\r
wait-serial SHORT_ OK
serial seq 1 75 > /tmp/uartsent; cat /tmp/uartsent >&3; dd bs=1 count=$(wc -c </tmp/uartsent) <&3 >/tmp/uartrecv 2>/dev/null; cmp /tmp/uartsent /tmp/uartrecv && echo "BURST_" "OK"\r
wait-serial BURST_ OK
quit
