# Orange Pi RV, the whole board with the mainline kernel (Alpine linux-stable 7.1 and the
# mainline jh7110-orangepi-rv.dtb): microSD card, Gigabit Ethernet, USB 3.0 and HDMI in
# one run. Mainline has no JH7110 display driver: HDMI is the framebuffer the boot firmware
# set up and handed over (simple-framebuffer, driven by simpledrm).
#   sbc-emu run --board orangepi-rv --firmware $F/fw_dynamic.bin --kernel $F/Image \
#       --dtb $F/jh7110-orangepi-rv.dtb --initrd $F/initramfs.cpio.gz --sd card.img --usb-drive stick.img \
#       --append "console=tty0 console=ttyS0,115200 earlycon rdinit=/init" --script scripts/orangepi-rv-board-mainline.txt
wait-serial orangepi-rv#
serial cat /proc/device-tree/model; echo; uname -r\n
wait-serial 7.1.5
# microSD
serial mount -t ext4 /dev/mmcblk1 /mnt && cat /mnt/hello.txt\n
wait-serial HELLO_FROM_HOST
serial dd if=/dev/urandom of=/mnt/blob bs=64k count=32 2>/dev/null; sha256sum /mnt/blob > /mnt/blob.sha; sync; echo 3 > /proc/sys/vm/drop_caches; sha256sum -c /mnt/blob.sha | sed 's/^/SD_/'\n
wait-serial SD_/mnt/blob: OK
# Gigabit Ethernet
serial ip link set eth0 up; sleep 3; echo $(cat /sys/class/net/eth0/speed /sys/class/net/eth0/carrier) LINK\n
wait-serial 1000 1 LINK
serial udhcpc -i eth0 -n -q -t 5 2>&1 | tail -1; ping -c 2 10.0.2.2 | grep received\n
wait-serial 2 packets received
serial wget -q -O - http://example.com/ | grep -o '<title>.*</title>'\n
wait-serial <title>Example Domain</title>
# USB 3.0
serial mkdir -p /media; mount -t ext4 /dev/sda /media && cat /media/hello.txt | sed 's/^/USB_/'\n
wait-serial USB_HELLO_FROM_HOST
# HDMI: the firmware framebuffer under simpledrm, with fbcon
serial cat /proc/fb; cat /sys/class/graphics/fb0/virtual_size | sed 's/^/FB_/'\n
wait-serial FB_1280,720
serial clear > /dev/tty1; echo 'Orange Pi RV (mainline 7.1): microSD, Gigabit Ethernet, USB 3.0 and HDMI' > /dev/tty1; ip -4 addr show eth0 | grep inet > /dev/tty1; df -h /mnt /media | tail -2 > /dev/tty1; echo SHOWN\n
wait-serial SHOWN
# a shell on the HDMI console, typed into on the USB keyboard
serial setsid sh -c 'exec sh -i </dev/tty1 >/dev/tty1 2>&1' & sleep 1; echo TTY1_""UP\n
wait-serial TTY1_UP
type echo typed on the USB keyboard > /tmp/kbd.txt; cat /tmp/kbd.txt\n
sleep 2
serial cat /tmp/kbd.txt | sed 's/^/KBD: /'\n
wait-serial KBD: typed on the USB keyboard
sleep 1
screenshot out/orangepi-rv-board-mainline.png
serial umount /mnt /media; echo BOARD_""OK\n
wait-serial BOARD_OK
quit
