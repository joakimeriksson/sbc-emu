# Orange Pi RV GPU (PowerVR BXE-4-32) with the open stack: mainline Linux's powervr driver
# loads the firmware file, boots the firmware (played by the emulator) and runtime-suspends
# the GPU; Mesa's PowerVR Vulkan driver finds the device, runs a compute shader
# and renders offscreen.
# Needs scripts/orangepi-rv-gpu.sh:
#   sbc-emu run --board orangepi-rv --firmware fw_dynamic.bin --kernel cache/orangepi-rv/gpu/Image \
#     --dtb jh7110-orangepi-rv.dtb --initrd cache/orangepi-rv/gpu/initramfs.cpio.gz --script ...
wait-serial orangepi-rv#
serial modprobe powervr exp_hw_support=1; dmesg | grep -E 'powervr|Initialized powervr' | sed 's/^.*\] //'\n
wait-serial Initialized powervr 1.0.0 for 18000000.gpu
serial sleep 1; cat /sys/class/drm/card1/device/power/runtime_status\n
wait-serial suspended
serial mount -t devtmpfs dev /debian/dev; mount -t proc proc /debian/proc; mount -t sysfs sys /debian/sys; echo MNT_""OK\n
wait-serial MNT_OK
serial PVR_I_WANT_A_BROKEN_VULKAN_DRIVER=1 chroot /debian vulkaninfo --summary 2>&1 | grep -E 'deviceName|driverName|deviceID'\n
wait-serial deviceName         = PowerVR B-Series BXE-4-32 MC1
# a Vulkan compute dispatch (tests/gpu/pvr/vkcompute.c): 4096 invocations, checked on the host
serial PVR_I_WANT_A_BROKEN_VULKAN_DRIVER=1 chroot /debian /usr/local/bin/vkcompute 2>&1 | grep VKCOMPUTE\n
wait-serial VKCOMPUTE_OK 4096 invocations
# offscreen rendering (tests/gpu/pvr/vktriangle.c): a cleared 64x64 RGBA8 target, then a
# triangle with red/green/blue corners, copied back to a buffer and checked on the host
serial PVR_I_WANT_A_BROKEN_VULKAN_DRIVER=1 chroot /debian /usr/local/bin/vktriangle clear 2>&1 | grep VKTRI\n
wait-serial VKTRI_OK clear
serial PVR_I_WANT_A_BROKEN_VULKAN_DRIVER=1 chroot /debian /usr/local/bin/vktriangle 2>&1 | grep -E 'corner|VKTRI'\n
wait-serial VKTRI_OK triangle
serial echo GPU_""DONE\n
wait-serial GPU_DONE
quit
