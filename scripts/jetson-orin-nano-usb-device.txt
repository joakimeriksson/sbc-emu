# The USB device-mode port on a full JetPack 6 image (docs/jetson-orin-nano.md, "USB device-mode
# port"): boot the image from --usb-drive with --usb-device on and log in (jetson/jetson, the
# image's documented default). Checks that tegra-xudc bound, the role switch is in device mode,
# NVIDIA's service configured the gadget, l4tbr0 has 192.168.55.1 and the PC took a DHCP lease.
wait-serial login:
serial jetson\r
wait-serial Password:
serial jetson\r
wait-serial $ 
serial for i in $(seq 120); do grep -q 192.168.55.100 /run/l4t-usb-devmode-dhcpd.leases 2>/dev/null && break; sleep 1; done; ls /sys/class/udc; cat /sys/class/usb_role/usb2-0-role-switch/role /sys/class/udc/*/state; systemctl is-active nv-l4t-usb-device-mode nv-l4t-usb-device-mode-runtime; ip -br -4 addr show l4tbr0\r
wait-serial 3550000.usb
wait-serial device
wait-serial configured
wait-serial active
wait-serial 192.168.55.1/24
serial grep -q 'lease 192.168.55.100' /run/l4t-usb-devmode-dhcpd.leases && echo "LEASE_" "OK"\r
wait-serial LEASE_ OK
quit
