# Raspberry Pi OS Lite (trixie, 2026-10-06) first boot on the emulated Pi 4, driven on the USB
# keyboard and HDMI console, ending with the browser's resume snapshot and a network check.
#   sbc-emu run --sd cache/os/raspios-lite.img --script scripts/raspios-firstboot.txt
# Dialogs flush pending input when they open, so every keystroke waits for the guest to go idle.
wait-serial raspberrypi login:
wait-idle 5
shot out/01-keyboard-layout.png
type \n
sleep 2
wait-idle 5
shot out/02-username.png
type pi\n
sleep 2
wait-idle 5
type raspberry\n
sleep 2
wait-idle 5
type raspberry\n
sleep 2
wait-idle 8
type \n
sleep 2
wait-idle 4
shot out/03-login-prompt.png
save out/raspios-lite.snap
type pi\n
sleep 2
wait-idle 4
type raspberry\n
sleep 3
wait-idle 6
type clear; sudo rfkill unblock wifi; sudo nmcli radio wifi on; sleep 8; sudo nmcli dev disconnect eth0\n
sleep 3
wait-idle 4
type raspberry\n
sleep 12
wait-idle 8
shot out/04-links.png
type nmcli dev wifi rescan; sleep 3; nmcli -f SSID,CHAN,SIGNAL,SECURITY dev wifi list\n
sleep 8
wait-idle 10
shot out/05-wifi-scan.png
type sudo nmcli dev wifi connect sbc-emu; sleep 5; ip -br -4 addr; ping -c 2 10.0.2.2; nmcli dev\n
sleep 15
wait-idle 15
shot out/06-wifi-connected.png
quit
